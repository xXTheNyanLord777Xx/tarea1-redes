import socket
import json

BUFF_SIZE = 50

class Message:
    def __init__(self, head, body, start_line):
        self.head = head
        self.body = body
        self.start_line = start_line

# Funcion que busca la url (target) en los dominios prohibidos
def contiene(url, prohibido):
    for dominio in prohibido:
        if dominio in url:
            return True
    return False

# Funcion que extrae todo lo que viene despues del dominio
def extraer(dominio,url):
    data = url.split(dominio,1)[1]
    print("data extraida post dominio: ", data)
    return data

# Funcion para recibir el mensaje sin importar el tamaño del buffsize
def recive_full_message(socket):
    response = b""
    first_chunk = socket.recv(BUFF_SIZE)
    response += first_chunk
    hasbody = True
    # Importante notar que estamos omitiendo que existe POST, de lo contrario el metodo falla
    if response.startswith(b"GET"):
        hasbody = False
    while True:
        if b"\r\n\r\n" in response:
            break
        chunk = socket.recv(BUFF_SIZE)
        response += chunk
    if hasbody:
        # Content Length es la longitud del body en bytes
        contentlength = (response.split(b"Content-Length: ")[1]).split(b"\r\n")[0]
        length = 0
        while length < int(contentlength):
            chunk = socket.recv(BUFF_SIZE)
            response += chunk
            length += BUFF_SIZE

    return response
    

# Funcion para transformar una secuencia de bytes a una estructura mas facil de trabajar
def parse_HTTP_message(http_message: bytes):

    particiones = http_message.split(b"\r\n\r\n")
    listhead = particiones[0].decode("utf-8").split("\r\n")
    start_line = listhead[0]
    head = {}
    for linea in listhead[1:]:
        data = linea.split(": ")
        head[data[0]] = data[1]
    body = particiones[1]
    return Message(head,body, start_line)

mensaje = parse_HTTP_message(mensajeoriginal)

# Funcion para transformar la estructura de mensaje en una secuencia de bytes que pueda ser interpretada por el cliente
def create_HTTP_message(message):
    head = message.head
    body = message.body
    data = message.start_line.encode("utf-8")
    data += b"\r\n"

    for key, value in head.items():
        data += key.encode("utf-8") + b": " + value.encode("utf-8")
        data += b"\r\n"

    data += b"\r\n"

    data += body

    return data
